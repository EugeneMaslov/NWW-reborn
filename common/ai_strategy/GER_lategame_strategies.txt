	reversed = yes
	allowed = { original_tag = GER }
	enable = {
		exists = yes
		is_subject = no
		NOT = { original_tag = GER }
		NOT = { has_government = fascism }
		NOT = { has_government = communism }
		has_war = yes
		any_enemy_country = { OR = { has_government = communism has_government = fascism } }
		NOT = { has_war_with = GER }
		GER = {
			has_focus_tree = GER_focus
			has_government = democratic
			is_subject = no
			has_completed_focus = GER_arsenal_of_democracy
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = send_volunteers_desire id = GER value = 200 }
	ai_strategy = { type = protect id = GER value = 200 }
	ai_strategy = { type = diplo_action_desire id = GER target = guarantee value = 200 }
}
