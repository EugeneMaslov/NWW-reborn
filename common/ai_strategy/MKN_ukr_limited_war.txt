# Owner MKN. WUW/TAOG pattern: common/ai_strategy/ITA.txt ITA_dont_call_allies,
# common/ai_strategy/SIA.txt VIC_dont_call_allies_in_limited_war.
# Reversed policies are evaluated for UKR's relations to each allied country.
UKR_mkn_do_not_call_allies = {
	allowed = { always = yes }
	enable = {
		OR = { is_in_faction_with = UKR is_ally_with = UKR }
		UKR = { has_country_flag = UKR_mkn_offensive_limited has_war_with = MKN }
	}
	abort_when_not_enabled = yes
	reversed = yes
	ai_strategy = { type = diplo_action_desire id = UKR target = call_allies value = -10000 }
}
UKR_mkn_allies_decline_limited_war = {
	allowed = { always = yes }
	enable = {
		NOT = { tag = UKR }
		UKR = { has_country_flag = UKR_mkn_offensive_limited has_war_with = MKN }
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = diplo_action_acceptance id = UKR target = call_allies value = -10000 }
	ai_strategy = { type = dont_join_wars_with id = UKR target_country = MKN value = 2000 }
}
