# Owner GER. Ready/target triggers are also evaluated by a native relation rule.
nww_GER_ai_campaign_ready = {
 has_war = no
 has_capitulated = no
 has_civil_war = no
 is_subject = no
 surrender_progress < 0.05
 has_stability > 0.40
 has_war_support > 0.35
 NOT = { has_country_flag = NWW_GER_campaign_recovery }
 NOT = { has_country_flag = NWW_GER_campaign_defeat_pending }
 NOT = { has_country_flag = NWW_GER_campaign_was_at_war }
}
nww_GER_ai_target_safe = {
 exists = yes
 is_subject = no
 NOT = { is_ally_with = GER }
 NOT = { has_war_with_major = yes }
 strength_ratio = { tag = GER ratio < 0.80 }
 NOT = { any_other_country = { has_guaranteed = PREV OR = { is_major = yes is_in_faction = yes is_subject = yes } } }
 OR = {
  is_in_faction = no
  AND = {
   OR = { is_faction_leader = yes is_major = no }
   NOT = { any_other_country = { is_major = yes is_in_faction_with = PREV NOT = { is_faction_leader = yes } } }
   NOT = { any_other_country = { is_faction_leader = yes is_in_faction_with = PREV NOT = { strength_ratio = { tag = GER ratio < 0.67 } } } }
  }
 }
}
