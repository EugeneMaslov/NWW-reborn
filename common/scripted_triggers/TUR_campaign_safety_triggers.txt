# Owner TUR. Ready/target triggers are also evaluated by a native relation rule.
nww_TUR_ai_campaign_ready = {
 has_war = no
 has_capitulated = no
 has_civil_war = no
 is_subject = no
 surrender_progress < 0.05
 has_stability > 0.40
 has_war_support > 0.35
 NOT = { has_country_flag = NWW_TUR_campaign_recovery }
 NOT = { has_country_flag = NWW_TUR_campaign_defeat_pending }
 NOT = { has_country_flag = NWW_TUR_campaign_was_at_war }
}
nww_TUR_ai_target_safe = {
 exists = yes
 is_subject = no
 NOT = { is_ally_with = TUR }
 is_major = no
 is_in_faction = no
 has_war = no
 strength_ratio = { tag = TUR ratio < 0.67 }
 NOT = { any_other_country = { has_guaranteed = PREV } }
}
# FROM is the proposed target, following ETH has_lon_arms_embargo.
# Use the explicit owner for readiness, independently of the calling event ROOT.
nww_TUR_ai_unsafe_declaration = {
 TUR = { is_ai = yes has_focus_tree = turkish_focus }
 OR = { TUR = { NOT = { nww_TUR_ai_campaign_ready = yes } } FROM = { NOT = { nww_TUR_ai_target_safe = yes } } }
}
