# Owner: GER. A foreign annexation/capitulation does not count as a German victory.
# Guarantor scope, PREV is Austria. The new refusal guarantee must raise the
# required preparation rather than permanently disabling the historical route.
GER_nww_containable_austrian_guarantor = {
	is_subject = no
	OR = { is_in_faction = no is_faction_leader = yes }
	OR = { tag = CZE tag = FRA }
	PREV = { tag = AUS }
	strength_ratio = { tag = GER ratio < 0.67 }
	GER = {
		has_focus_tree = GER_focus
		date > 1940.12.31
		OR = {
			AND = { has_government = neutrality has_completed_focus = GER_gott_mit_uns }
			AND = { has_government = fascism has_completed_focus = GER_the_new_reich }
		}
	}
}
GER_nww_eastern_campaigns_settled = {
	OR = {
		AND = { POL = { exists = no } owns_state = 10 }
		POL = { exists = yes is_subject_of = PREV }
		POL = { exists = yes has_capitulated = yes has_war_with = PREV }
	}
	OR = {
		AND = { AUS = { exists = no } owns_state = 4 }
		AUS = { exists = yes is_subject_of = PREV }
		AUS = { exists = yes has_capitulated = yes has_war_with = PREV }
	}
}
