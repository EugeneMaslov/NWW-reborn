nww_TUR_random_ai_mode = {
	OR = {
		has_game_rule = {
			rule = TUR_ai_behavior
			option = RANDOM
		}
		AND = {
			is_historical_focus_on = no
			NOT = {
				OR = {
					has_game_rule = {
						rule = TUR_ai_behavior
						option = HISTORICAL
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = OTTOMAN
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = ALTERNATE_KEMALIST
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = BALKAN_ENTENTE
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = TUR_DEMOCRATIC_KEMALIST
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = TUR_DEMOCRATIC_ALTERNATE
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = COMMUNIST_TUR_AI
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = TUR_COMMUNISM_ALTERNATE
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = FASCISM_TUR_AI
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = TUR_FASCISM_ALTERNATE
					}
				}
			}
		}
	}
}

			NOT = {
				OR = {
					has_game_rule = {
						rule = TUR_ai_behavior
						option = HISTORICAL
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = OTTOMAN

						option = ALTERNATE_KEMALIST

						option = BALKAN_ENTENTE
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = TUR_DEMOCRATIC_KEMALIST
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = TUR_DEMOCRATIC_ALTERNATE
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = COMMUNIST_TUR_AI
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = TUR_COMMUNISM_ALTERNATE
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = FASCISM_TUR_AI
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = TUR_FASCISM_ALTERNATE
					}
					has_game_rule = {
						rule = TUR_ai_behavior
						option = RANDOM
					}
				}
			}
		}
		AND = {
			nww_TUR_random_ai_mode = yes
			OR = {
				has_country_flag = TUR_AI_RANDOM_NEUTRALITY_HISTORICAL
				has_country_flag = TUR_AI_RANDOM_OTTOMAN
			}
		}
	}
}
