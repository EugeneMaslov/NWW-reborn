on_actions = {
	on_startup = {
		effect = {
			SOV = {
				if = {
					limit = {
						has_focus_tree = SOV_focus
					}
					nww_SOV_suspend_detached_purges = yes
					nww_SOV_update_reconstruction = yes
					if = {
						limit = {
							has_completed_focus = NWW_SOV_litbel_campaign
						}
						nww_SOV_prepare_litbel_campaign = yes
					}
					if = {
						limit = {
							has_completed_focus = NWW_SOV_steppe_campaign
						}
						nww_SOV_prepare_steppe_campaign = yes
					}
				}
			}
		}
	}
	on_daily_SOV = {
		effect = {
			if = {
				limit = {
					has_focus_tree = SOV_focus
				}
				nww_SOV_suspend_detached_purges = yes
				nww_SOV_update_reconstruction = yes
				if = {
					limit = {
						has_completed_focus = NWW_SOV_litbel_campaign
					}
					nww_SOV_prepare_litbel_campaign = yes
				}
				if = {
					limit = {
						has_completed_focus = NWW_SOV_steppe_campaign
					}
					nww_SOV_prepare_steppe_campaign = yes
				}
			}
		}
	}
}
