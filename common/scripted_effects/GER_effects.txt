nww_GER_initialize_politics = {
	if = {
		limit = {
			OR = {
				has_completed_focus = GER_abolish_the_hohenzollerns
				has_country_flag = ger_spartacist_side
				has_country_flag = ger_bop_removed
			}
		}
		set_country_flag = ger_bop_removed
		if = {
			limit = {
				has_power_balance = {
					id = GER_kaiser_vs_reichstag
				}
			}
			remove_power_balance = {
				id = GER_kaiser_vs_reichstag
			}
		}
	}
	else = {
		if = {
			limit = {
				NOT = {
					has_power_balance = {
						id = GER_kaiser_vs_reichstag
					}
				}
			}
			set_power_balance = {
				id = GER_kaiser_vs_reichstag
				set_default = yes
			}
		}
		if = {
			limit = {
				OR = {
					has_completed_focus = GER_parliamentary_spring
					has_country_flag = ger_parliamentary_spring
				}
			}
			set_country_flag = ger_bop_frozen
			if = {
				limit = {
					NOT = {
						power_balance_value = {
							id = GER_kaiser_vs_reichstag
							value = 1
						}
					}
				}
				set_power_balance = {
					id = GER_kaiser_vs_reichstag
					set_value = 1
				}
			}
		}
		else_if = {
			limit = {
				has_completed_focus = GER_dissolve_the_reichstag
			}
			set_country_flag = ger_bop_frozen
			if = {
				limit = {
					NOT = {
						power_balance_value = {
							id = GER_kaiser_vs_reichstag
							value = -1
						}
					}
				}
				set_power_balance = {
					id = GER_kaiser_vs_reichstag
					set_value = -1
				}
			}
		}
	}
	nww_GER_eastern_war_guard = yes
	nww_GER_set_party_labels = yes
}

nww_GER_mitteleuropa_demand = {
	if = {
		limit = {
			exists = yes
			NOT = {
				has_war_with = GER
			}
			GER = {
				exists = yes
				is_subject = no
				has_government = neutrality
			}
			date > 1938.12.31
		}
		if = {
			limit = {
				is_subject_of = GER
				OR = {
					AND = { has_dlc = "Together for Victory" NOT = { has_autonomy_state = autonomy_integrated_puppet } }
					AND = { NOT = { has_dlc = "Together for Victory" } NOT = { has_autonomy_state = autonomy_puppet } }
				}
			}
			if = {
				limit = {
					has_dlc = "Together for Victory"
				}
				GER = {
					set_autonomy = {
						target = PREV
						autonomy_state = autonomy_integrated_puppet
						end_wars = no
						end_civil_wars = no
					}
				}
			}
			else = {
				GER = {
					set_autonomy = {
						target = PREV
						autonomy_state = autonomy_puppet
						end_wars = no
						end_civil_wars = no
					}
				}
			}
		}
		else_if = {
			limit = {
				is_subject = no
				NOT = { has_country_flag = GER_mitteleuropa_demand_sent }
				NOT = { has_country_flag = ger_mitteleuropa_refused }
			}
			set_country_flag = GER_mitteleuropa_demand_sent
			country_event = {
				id = ger_crisis.20
				days = 1
			}
		}
	}
}
