nww_mkn_initialize_bop = {
 if = {
  limit = { has_country_flag = MKN_crisis_enabled }
  if = {
   limit = { OR = { has_completed_focus = MKN_abolish_election_of_commanders has_completed_focus = MKN_sever_urban_ties has_completed_focus = MKN_embassy_to_kharkov has_country_flag = MKN_collapse_started } }
   set_country_flag = MKN_bop_abandoned
   if = { limit = { has_power_balance = { id = MKN_power_balance } } remove_power_balance = { id = MKN_power_balance } }
  }
  else_if = {
   limit = { has_completed_focus = MKN_abolish_general_staff }
   set_country_flag = MKN_bop_locked
   if = { limit = { NOT = { power_balance_value = { id = MKN_power_balance value = 1 } } } set_power_balance = { id = MKN_power_balance set_value = 1 } }
  }
  else_if = {
   limit = { has_completed_focus = MKN_congress_1936 NOT = { has_country_flag = MKN_bop_abandoned } NOT = { has_power_balance = { id = MKN_power_balance } } }
   set_power_balance = { id = MKN_power_balance set_value = 0 }
  }
 }
 else_if = {
  limit = { has_completed_focus = MKN_nestor_the_first }
  if = { limit = { has_power_balance = { id = MKN_power_balance } } remove_power_balance = { id = MKN_power_balance } }
  set_country_flag = MKN_bop_abandoned
 }
 else_if = {
  limit = { NOT = { has_country_flag = MKN_bop_abandoned } NOT = { has_power_balance = { id = MKN_power_balance } } }
  set_power_balance = { id = MKN_power_balance set_value = 0 }
  if = { limit = { has_completed_focus = MKN_black_hundreds } set_power_balance = { id = MKN_power_balance set_value = -0.65 } }
  else_if = { limit = { OR = { has_completed_focus = MKN_constituent_assembly has_completed_focus = MKN_everlasting_makhnovshchina has_completed_focus = MKN_council_communism } } set_power_balance = { id = MKN_power_balance set_value = 0.65 } }
  else_if = { limit = { has_completed_focus = MKN_red_banner_steppe } set_power_balance = { id = MKN_power_balance set_value = 0.35 } }
 }
}
