# Owner GER. Explicit AI rules override historical mode. Select once per campaign.
GER_nww_initialize_ai_route = {
 if = {
  limit = { is_ai = yes has_focus_tree = GER_focus NOT = { has_country_flag = GER_nww_ai_route_selected } }
  if = { limit = { has_completed_focus = GER_the_potsdam_coup } set_country_flag = GER_nww_ai_authoritarian }
  else_if = { limit = { has_completed_focus = GER_parliamentary_spring } set_country_flag = GER_nww_ai_democratic }
  else_if = { limit = { has_completed_focus = GER_the_pan_germanic_triumph } set_country_flag = GER_nww_ai_fascist }
  else_if = { limit = { has_completed_focus = GER_the_spartacist_uprising } set_country_flag = GER_nww_ai_communist }
  else_if = { limit = { OR = { has_game_rule = { rule = GER_ai_behavior option = KAISER } } } set_country_flag = GER_nww_ai_authoritarian }
  else_if = { limit = { OR = { has_game_rule = { rule = GER_ai_behavior option = DEMOCRATIC } } } set_country_flag = GER_nww_ai_democratic }
  else_if = { limit = { OR = { has_game_rule = { rule = GER_ai_behavior option = FASCIST } has_game_rule = { rule = GER_ai_behavior option = GER_ALT_FASCIST } } } set_country_flag = GER_nww_ai_fascist }
  else_if = { limit = { OR = { has_game_rule = { rule = GER_ai_behavior option = GER_COMMUNIST_STALINIST } has_game_rule = { rule = GER_ai_behavior option = GER_COMMUNIST_SPARTACUS } } } set_country_flag = GER_nww_ai_communist }
  else_if = { limit = { has_game_rule = { rule = GER_ai_behavior option = DEFAULT } is_historical_focus_on = yes } set_country_flag = GER_nww_ai_authoritarian }
  else = { random_list = { seed = random
   25 = { set_country_flag = GER_nww_ai_authoritarian }
   25 = { set_country_flag = GER_nww_ai_democratic }
   25 = { set_country_flag = GER_nww_ai_fascist }
   25 = { set_country_flag = GER_nww_ai_communist }
  } }
  set_country_flag = GER_nww_ai_route_selected
 }
}
