POL_initialize_country = {
	if = {
		limit = {
			NOT = {
				has_focus_tree = POL_focus
			}
		}
		load_focus_tree = {
			tree = POL_focus
			keep_completed = yes
		}
	}
	if = {
		limit = {
			NOT = {
				has_country_flag = POL_nww_initialized
			}
		}
		POL_cleanup_native_subsystems = yes
		if = {
			limit = {
				NOT = {
					has_country_flag = POL_nww_start_configured
				}
			}
			if = {
				limit = {
					NOT = {
						has_country_flag = POL_german_dependence_ended
					}
					NOT = {
						has_completed_focus = POL_economic_independence
					}
					NOT = {
						has_completed_focus = POL_nationalize_german_assets
					}
				}
				add_ideas = POL_german_economic_stranglehold
			}
			if = {
				limit = {
					NOT = {
						has_country_flag = POL_crisis_resolved
					}
				}
				add_ideas = POL_sanation_crisis
			}
			set_country_flag = POL_nww_start_configured
		}
		POL_install_campaign_guard = yes
		set_country_flag = POL_nww_initialized
	}
	if = {
		limit = {
			has_completed_focus = POL_the_colonels_regime
		}
		set_country_flag = POL_path_sanation
	}
	if = {
		limit = {
			has_completed_focus = POL_endecja_ascendant
		}
		set_country_flag = POL_path_endecja
	}
	if = {
		limit = {
			has_completed_focus = POL_pragmatic_deal
		}
		set_country_flag = POL_path_german
	}
	if = {
		limit = {
			has_completed_focus = POL_the_lodz_uprising
		}
		set_country_flag = POL_path_communist
	}
	if = {
		limit = {
			OR = {
				has_country_flag = POL_german_dependence_ended
				has_completed_focus = POL_economic_independence
				has_completed_focus = POL_nationalize_german_assets
				has_completed_focus = POL_wartime_conversion
				has_war_with = GER
			}
		}
		POL_end_german_dependence = yes
	}
	else_if = {
		limit = {
			has_completed_focus = POL_accept_german_capital
			NOT = {
				has_idea = POL_german_economic_stranglehold_2
			}
		}
		POL_accept_capital = yes
	}
	if = {
		limit = {
			NOT = {
				has_country_flag = POL_campaign_guard_installed
			}
		}
		POL_install_campaign_guard = yes
	}
	if = {
		limit = {
			is_ai = yes
			NOT = {
				has_country_flag = POL_ai_selected
			}
		}
		POL_select_ai_path = yes
	}
	POL_refresh_crisis = yes
	POL_refresh_programs = yes
	POL_refresh_coast = yes
	if = {
		limit = {
			has_completed_focus = POL_march_to_the_sea
		}
		POL_prepare_western_war = yes
	}
}
POL_select_ai_path = {
 if = { limit = { has_country_flag = POL_path_sanation } set_country_flag = POL_ai_sanation }
 else_if = { limit = { has_country_flag = POL_path_endecja } set_country_flag = POL_ai_endecja }
 else_if = { limit = { has_country_flag = POL_path_german } set_country_flag = POL_ai_german }
 else_if = { limit = { has_country_flag = POL_path_communist } set_country_flag = POL_ai_communist }
 else_if = {
		limit = {
			has_game_rule = {
				rule = POL_ai_behavior
				option = COMMUNIST
			}
		}
		set_country_flag = POL_ai_communist
	}
	else_if = {
		limit = {
			has_game_rule = {
				rule = POL_ai_behavior
				option = FASCIST_INDEPENDENT
			}
		}
		set_country_flag = POL_ai_endecja
	}
	else_if = {
		limit = {
			has_game_rule = {
				rule = POL_ai_behavior
				option = FASCIST
			}
		}
		set_country_flag = POL_ai_german
	}
	else_if = {
		limit = {
			OR = {
				has_game_rule = {
					rule = POL_ai_behavior
					option = HISTORICAL
				}
				AND = {
					is_historical_focus_on = yes
					has_game_rule = {
						rule = POL_ai_behavior
						option = DEFAULT
					}
				}
			}
		}
		set_country_flag = POL_ai_sanation
	}
	else = {
		random_list = {
			35 = {
				set_country_flag = POL_ai_sanation
			}
			25 = {
				set_country_flag = POL_ai_endecja
			}
			20 = {
				set_country_flag = POL_ai_german
			}
			20 = {
				set_country_flag = POL_ai_communist
			}
		}
	}
	set_country_flag = POL_ai_selected
}
