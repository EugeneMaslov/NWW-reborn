# Owner TUR. Negative relation rules combine without clearing other protections.
nww_TUR_ai_install_campaign_guard = {
 remove_relation_rule_override = { trigger = nww_TUR_ai_unsafe_declaration can_not_declare_war = yes }
 add_relation_rule_override = { trigger = nww_TUR_ai_unsafe_declaration usage_desc = NWW_TUR_campaign_safety_tt can_not_declare_war = yes }
 set_country_flag = NWW_TUR_campaign_guard_installed
}
nww_TUR_ai_observe_campaign = {
 if = { limit = { NOT = { has_country_flag = NWW_TUR_campaign_guard_installed } } nww_TUR_ai_install_campaign_guard = yes }
 if = { limit = { has_capitulated = yes } set_country_flag = NWW_TUR_campaign_defeat_pending }
 if = { limit = { has_war = yes } set_country_flag = NWW_TUR_campaign_was_at_war }
 if = {
  limit = { has_war = no OR = { has_country_flag = NWW_TUR_campaign_was_at_war has_country_flag = NWW_TUR_campaign_defeat_pending } }
  if = {
   limit = { has_country_flag = NWW_TUR_campaign_defeat_pending }
   set_country_flag = { flag = NWW_TUR_campaign_recovery days = 730 }
  }
  else = { set_country_flag = { flag = NWW_TUR_campaign_recovery days = 180 } }
  clr_country_flag = NWW_TUR_campaign_was_at_war
  clr_country_flag = NWW_TUR_campaign_defeat_pending
 }
}
