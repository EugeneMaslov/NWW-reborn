		remove_ideas = POL_german_economic_stranglehold
		add_ideas = POL_german_economic_stranglehold_2
	}
}

POL_refresh_crisis = {
	if = {
		limit = {
			OR = {
				has_country_flag = POL_crisis_resolved
				has_completed_focus = POL_professional_civil_service
				has_completed_focus = POL_national_bloc
				has_completed_focus = POL_customs_commission
				has_completed_focus = POL_council_republic
			}
		}
		POL_finish_crisis = yes
	}
	else_if = {
		limit = { OR = { has_completed_focus = POL_state_budget_commission has_country_flag = POL_budget_commission_done } has_idea = POL_sanation_crisis }
		swap_ideas = { remove_idea = POL_sanation_crisis add_idea = POL_sanation_crisis_2 }
	}
}

# A real civil war, following GER's installed/reviewed player transfer pattern.
POL_start_communist_civil_war = {
	set_country_flag = POL_path_communist
	set_country_flag = POL_sanation_loyalists
	start_civil_war = {
		ideology = communism
		size = 0.40
		set_country_flag = POL_path_communist
		set_country_flag = POL_communist_rebels
		clr_country_flag = POL_sanation_loyalists
		set_cosmetic_tag = POL
		if = { limit = { is_in_faction = yes } leave_faction = yes }
		load_focus_tree = { tree = POL_focus keep_completed = yes copy_completed_from = ROOT }
		set_politics = { ruling_party = communism elections_allowed = no election_frequency = 48 }
		ROOT = { POL_maria_koszutska = { set_nationality = PREV.PREV } }
		POL_maria_koszutska = { add_country_leader_role = { country_leader = { ideology = marxism expire = 1965.1.1 traits = { } } promote_leader = yes } }
		POL_refresh_programs = yes
		POL_install_campaign_guard = yes
		change_tag_from = ROOT
	}
}
POL_install_campaign_guard = {
	if = {
		limit = {
			is_ai = yes
			OR = {
				date < 1939.1.1
				NOT = { has_country_flag = POL_path_endecja }
				NOT = { has_completed_focus = POL_march_to_the_sea }
				NOT = { POL_can_prepare_western_war = yes }
			}
		}
		set_rule = { can_not_declare_war = yes }
		set_country_flag = POL_declaration_rule_owned
	}
	else_if = {
		limit = { has_country_flag = POL_declaration_rule_owned }
		clear_rule = { can_not_declare_war = yes }
		clr_country_flag = POL_declaration_rule_owned
	}
	set_country_flag = POL_campaign_guard_installed
}
