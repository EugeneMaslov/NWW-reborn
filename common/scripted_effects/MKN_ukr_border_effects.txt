mkn_ukr_initialize_frontier = {
	if = {
		limit = {
			NOT = {
				has_country_flag = mkn_ukr_frontier_initialized
			}
		}
		set_variable = {
			mkn_ukr_escalation_level = 1
		}
		set_variable = {
			mkn_consecutive_border_wars = 0
		}
		set_country_flag = mkn_ukr_frontier_initialized
	}
	if = {
		limit = {
			has_completed_focus = MKN_ukrainian_frontier_root
			NOT = {
				has_country_flag = mkn_ukr_frontier_concluded
			}
			NOT = {
				has_country_flag = mkn_ukr_frontier_concluded
			}
			NOT = {
				has_country_flag = mkn_ukr_frontier_concluded
			}
			NOT = {
				has_country_flag = mkn_ukr_frontier_concluded
			}
		}
		set_country_flag = mkn_ukr_frontier_unlocked
	}
	if = {
		limit = {
			has_completed_focus = MKN_tachanka_deep_recon
		}
		set_country_flag = mkn_ukr_recon_discount
	}
	clamp_variable = {
		var = mkn_ukr_escalation_level
		min = 1
		max = 5
	}
	mkn_ukr_update_decision_costs = yes
}
mkn_ukr_check_escalation = {
	if = {
		limit = {
			NOT = {
				has_country_flag = mkn_ukr_frontier_concluded
			}
			OR = {
				check_variable = {
					var = mkn_ukr_escalation_level
					value = 5
					compare = greater_than_or_equals
				}
				check_variable = {
					var = mkn_consecutive_border_wars
					value = 3
					compare = greater_than_or_equals
				}
			}
		}
		set_variable = {
			mkn_ukr_escalation_level = 5
		}
		set_country_flag = mkn_ukr_war_pending
	}
}
mkn_ukr_record_border_battle = {
	if = {
		limit = {
			NOT = {
				has_country_flag = mkn_ukr_frontier_concluded
			}
		}
		if = {
			limit = {
				NOT = {
					has_country_flag = mkn_ukr_recent_border_battle
				}
			}
			set_variable = {
				mkn_consecutive_border_wars = 0
			}
		}
		add_to_variable = {
			mkn_consecutive_border_wars = 1
		}
		set_country_flag = {
			flag = mkn_ukr_recent_border_battle
			days = 180
		}
		set_variable = {
			mkn_ukr_escalation_level = 1
		}
		clr_country_flag = mkn_ukr_border_battle_active
		mkn_ukr_check_escalation = yes
	}
}

mkn_ukr_border_victory = {
	if = {
		limit = {
			tag = MKN
			NOT = {
				has_country_flag = mkn_ukr_frontier_concluded
			}
			FROM.FROM = {
				OR = {
					tag = UKR
					tag = SOV
				}
			}
			FROM.FROM.FROM = {
				NOT = {
					state = 202
				}
				OR = {
					AND = {
						FROM.FROM = {
							tag = UKR
						}
						OR = {
							is_owned_by = UKR
							is_owned_by = MKN
						}
					}
					AND = {
						FROM.FROM = {
							tag = SOV
						}
						OR = {
							is_owned_by = SOV
							is_owned_by = MKN
						}
						OR = {
							state = 227
							state = 221
							state = 137
						}
						MKN = {
							has_completed_focus = MKN_donbas_miners_agitation
						}
					}
				}
			}
		}
		FROM.FROM.FROM = {
			mkn_ukr_integrate_won_state = yes
			MKN = {
				set_variable = {
					mkn_ukr_last_contested_state = PREV
				}
			}
		}
		custom_effect_tooltip = MKN_frontier_core_integration_tt
		add_war_support = 0.10
		set_country_flag = mkn_ukr_last_battle_won
		mkn_ukr_record_border_battle = yes
		country_event = {
			id = mkn_ukr_border.2
		}
	}
}

mkn_ukr_frontier_daily = {
	mkn_ukr_initialize_frontier = yes
	if = {
		limit = {
			NOT = {
				has_country_flag = mkn_ukr_frontier_concluded
			}
			OR = {
				has_country_flag = MKN_collapse_started
				NOT = {
					UKR = {
						exists = yes
					}
				}
			}
		}
		set_country_flag = mkn_ukr_frontier_concluded
		set_country_flag = mkn_ukr_frontier_aborted
		clr_country_flag = mkn_frontier_year_running
		remove_mission = MKN_frontier_year_countdown_mission
		clr_country_flag = mkn_ukr_pact_pending
		clr_country_flag = mkn_ukr_war_pending
	}
	if = {
		limit = {
			OR = {
				has_war_with = UKR
				has_war_with = SOV
			}
		}
		set_country_flag = mkn_ukr_frontier_concluded
		mkn_ukr_stop_border_battle = yes
		set_country_flag = mkn_ukr_total_war
		clr_country_flag = mkn_frontier_year_running
		remove_mission = MKN_frontier_year_countdown_mission
		set_variable = {
			mkn_ukr_escalation_level = 5
		}
		clr_country_flag = mkn_ukr_pact_pending
		clr_country_flag = mkn_ukr_war_pending
		if = {
			limit = {
				NOT = {
					has_idea = MKN_war_for_the_steppe
				}
			}
			add_ideas = MKN_war_for_the_steppe
		}
	}
	if = {
		limit = {
			NOT = {
				has_country_flag = mkn_ukr_frontier_concluded
			}
			has_border_war_with = UKR
		}
		set_country_flag = {
			flag = mkn_ukr_recent_border_battle
			days = 180
		}
	}
	else_if = {
		limit = {
			NOT = {
				has_country_flag = mkn_ukr_recent_border_battle
			}
		}
		set_variable = {
			mkn_consecutive_border_wars = 0
		}
	}
	if = {
		limit = {
			has_country_flag = mkn_ukr_border_battle_active
			NOT = {
				has_border_war_with = UKR
			}
		}
		clr_country_flag = mkn_ukr_border_battle_active
	}
	if = {
		limit = {
			has_country_flag = mkn_ukr_pact_pending
			NOT = {
				mkn_ukr_pact_context_valid = yes
			}
		}
		clr_country_flag = mkn_ukr_pact_pending
	}
	if = {
		limit = {
			has_country_flag = UKR_MKN_border_settled
			NOT = {
				has_country_flag = mkn_ukr_frontier_concluded
			}
		}
		set_country_flag = mkn_ukr_frontier_concluded
		set_country_flag = mkn_ukr_pact_signed
		clr_country_flag = mkn_ukr_war_pending
	}
	if = {
		limit = {
			has_country_flag = mkn_ukr_frontier_pact_owned
			NOT = {
				has_country_flag = mkn_ukr_frontier_pact_term
			}
		}
		diplomatic_relation = {
			country = UKR
			relation = non_aggression_pact
			active = no
		}
		clr_country_flag = mkn_ukr_frontier_pact_owned
	}
	mkn_ukr_check_escalation = yes
	if = {
		limit = {
			has_country_flag = mkn_ukr_war_pending
			NOT = {
				has_border_war_with = UKR
			}
			UKR = {
				exists = yes
			}
			NOT = {
				is_ally_with = UKR
			}
		}
		country_event = {
			id = mkn_ukr_border.1
		}
	}
	if = {
		limit = {
			has_country_flag = mkn_ukr_frontier_concluded
		}
		remove_mission = MKN_passive_escalation_drift
		clr_country_flag = mkn_ukr_frontier_unlocked
		clr_country_flag = mkn_frontier_conflict_active
	}
}
mkn_ukr_stop_border_battle = {
	if = {
		limit = {
			OR = {
				OR = {
					OR = {
						OR = {
							OR = {
								has_border_war_with = UKR
								has_border_war_with = SOV
							}
							has_border_war_with = SOV
						}
						has_border_war_with = SOV
					}
					has_border_war_with = SOV
				}
				has_border_war_with = SOV
			}
		}
		if = {
			limit = {
				has_variable = mkn_ukr_attacking_state
				has_variable = mkn_ukr_defending_state
			}
			cancel_border_war = {
				attacker = var:mkn_ukr_attacking_state
				defender = var:mkn_ukr_defending_state
				dont_fire_events = yes
			}
		}
		else = {
			every_owned_state = {
				limit = {
					is_controlled_by = ROOT
					has_border_war = yes
				}
				MKN = {
					set_variable = {
						mkn_ukr_attacking_state = PREV
					}
				}
				every_neighbor_state = {
					limit = {
						is_owned_and_controlled_by = UKR
						has_border_war = yes
					}
					MKN = {
						set_variable = {
							mkn_ukr_defending_state = PREV
						}
						cancel_border_war = {
							attacker = var:mkn_ukr_attacking_state
							defender = var:mkn_ukr_defending_state
							dont_fire_events = yes
						}
					}
				}
			}
		}
	}
	clr_country_flag = mkn_ukr_border_battle_active
}
