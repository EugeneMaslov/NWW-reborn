# NWW owner: GER. Remove only the obsolete native Axis template; preserve
# Mitteleuropa, accepted Soviet alliances and every other diplomatic choice.
nww_GER_remove_starting_axis = {
	if = {
		limit = {
			has_faction_template = faction_template_axis
			# Earlier saves could rename their original Axis into Mitteleuropa
			# without replacing the template. Preserve that completed diplomacy.
			NOT = {
				is_faction_leader = yes
				OR = {
					has_completed_focus = GER_mitteleuropa_economic_community
					AND = {
						has_completed_focus = GER_the_austrian_referendum
						AUS = { is_in_faction_with = PREV }
					}
				}
			}
		}
		if = { limit = { is_faction_leader = yes } dismantle_faction = yes }
		else = { leave_faction = yes }
	}
}
