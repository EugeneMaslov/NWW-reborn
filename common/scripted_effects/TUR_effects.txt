nww_TUR_update_imperial_crisis = {
	if = {
		limit = {
			has_country_flag = NWW_TUR_council_settled
		}
		if = {
			limit = {
				has_idea = NWW_TUR_imperial_crisis
			}
			remove_ideas = NWW_TUR_imperial_crisis
		}
		if = {
			limit = {
				has_idea = NWW_TUR_imperial_crisis_1
			}
			remove_ideas = NWW_TUR_imperial_crisis_1
		}
		if = {
			limit = {
				has_idea = NWW_TUR_imperial_crisis_2
			}
			remove_ideas = NWW_TUR_imperial_crisis_2
		}
		if = {
			limit = {
				NOT = {
					has_idea = NWW_TUR_imperial_compact
				}
			}
			add_ideas = NWW_TUR_imperial_compact
		}
	}
	else_if = {
		limit = {
			OR = {
				AND = {
					has_country_flag = NWW_TUR_officer_accord
					has_country_flag = NWW_TUR_ulema_accord
				}

					has_country_flag = NWW_TUR_officer_accord

				}

nww_TUR_prepare_bul_campaign = {
	if = {
		limit = {
			tag = TUR
			is_subject = no
			BUL = {
				exists = yes
			}
			NOT = {
				is_ally_with = BUL
			}
		}
		add_state_claim = 48
		add_state_claim = 211
		add_state_claim = 212
		add_state_claim = 801
		BUL = {
			every_owned_state = {
				add_claim_by = ROOT
			}
		}
		if = {
			limit = {
				NOT = {
					has_war_with = BUL
				}
				NOT = {
					has_wargoal_against = BUL
				}
			}
			if = {
				limit = {
					nww_TUR_ai_can_target_BUL = yes
				}
				create_wargoal = {
					type = annex_everything
					target = BUL
				}
			}
		}
		add_state_claim = 1083
		add_state_claim = 184
	}
}
