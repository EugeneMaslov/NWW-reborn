# Owner GER. The engine rejects declaration bans in relation overrides.
# Use an owned country rule while unready, plus per-target AI strategies.
nww_GER_ai_install_campaign_guard = {
 set_country_flag = NWW_GER_campaign_guard_installed
}
nww_GER_ai_refresh_declaration_rule = {
 if = {
  limit = { is_ai = yes OR = { NOT = { nww_GER_ai_campaign_ready = yes } date < 1939.1.1 } }
  set_rule = { can_not_declare_war = yes }
  set_country_flag = NWW_GER_declaration_rule_owned
 }
 else_if = {
  limit = { has_country_flag = NWW_GER_declaration_rule_owned }
  clear_rule = { can_not_declare_war = yes }
  clr_country_flag = NWW_GER_declaration_rule_owned
 }
}
nww_GER_ai_observe_campaign = {
 if = { limit = { NOT = { has_country_flag = NWW_GER_campaign_guard_installed } } nww_GER_ai_install_campaign_guard = yes }
 if = { limit = { has_capitulated = yes } set_country_flag = NWW_GER_campaign_defeat_pending }
 if = { limit = { has_war = yes } set_country_flag = NWW_GER_campaign_was_at_war }
 if = {
  limit = { has_war = no OR = { has_country_flag = NWW_GER_campaign_was_at_war has_country_flag = NWW_GER_campaign_defeat_pending } }
  if = {
   limit = { has_country_flag = NWW_GER_campaign_defeat_pending }
   set_country_flag = { flag = NWW_GER_campaign_recovery days = 730 }
  }
  else = { set_country_flag = { flag = NWW_GER_campaign_recovery days = 180 } }
  clr_country_flag = NWW_GER_campaign_was_at_war
  clr_country_flag = NWW_GER_campaign_defeat_pending
 }
 nww_GER_ai_refresh_declaration_rule = yes
}
