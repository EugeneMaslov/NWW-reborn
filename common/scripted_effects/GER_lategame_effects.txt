GER_nww_partial_army_reform = {
	if = {
		limit = { has_idea = GER_imperial_overstretch }
		swap_ideas = {
			remove_idea = GER_imperial_overstretch
			add_idea = GER_imperial_overstretch_2
		}
	}
}
# The old communist focus could complete while campaign safety prevented issuing
# its goals. Recheck after 1941 and recovery, including expired late-game goals.
GER_nww_refresh_western_wargoals = {
	if = {
		limit = { GER_nww_western_mandate = yes }
		if = {
			limit = {
				FRA = { exists = yes is_subject = no }
				NOT = { is_ally_with = FRA }
				NOT = { has_war_with = FRA }
				NOT = { has_wargoal_against = FRA }
				nww_GER_ai_can_target_FRA = yes
			}
			create_wargoal = { type = annex_everything target = FRA }
		}
		if = {
			limit = {
				OR = { has_completed_focus = GER_world_revolution has_completed_focus = GER_the_second_naval_race }
				ENG = { exists = yes is_subject = no }
				NOT = { is_ally_with = ENG }
				NOT = { has_war_with = ENG }
				NOT = { has_wargoal_against = ENG }
				nww_GER_ai_can_target_ENG = yes
			}
			create_wargoal = { type = annex_everything target = ENG }
		}
	}
}
# Repair only existing debuffs. Never recreate one after a capstone or repeat rewards.
GER_nww_repair_army_reform = {
	if = {
		limit = {
			OR = { has_idea = GER_imperial_overstretch has_idea = GER_imperial_overstretch_2 }
			OR = {
				has_completed_focus = GER_gott_mit_uns
				has_completed_focus = GER_arsenal_of_democracy
				has_completed_focus = GER_the_new_reich
				has_completed_focus = GER_workers_militias
			}
		}
		remove_ideas = GER_imperial_overstretch
		remove_ideas = GER_imperial_overstretch_2
	}
	else_if = {
		limit = {
			has_idea = GER_imperial_overstretch
			OR = {
				has_completed_focus = GER_fund_the_reichswehr
				has_completed_focus = GER_true_constitutional_monarchy
				has_completed_focus = GER_abolish_the_hohenzollerns
				has_completed_focus = GER_german_soviet_republic
			}
		}
		GER_nww_partial_army_reform = yes
	}
}
# Called in AUS scope after the refusal's live diplomacy guard. Prefer CZE, fall back to FRA.
GER_nww_austrian_guarantee = {
	if = {
		limit = {
			CZE = {
				exists = yes
				is_subject = no
				NOT = { has_war_with = GER }
				NOT = { has_war_with = AUS }
				NOT = { is_ally_with = GER }
			}
		}
		CZE = { give_guarantee = AUS }
	}
	else_if = {
		limit = {
			FRA = {
				exists = yes
				is_subject = no
				NOT = { has_war_with = GER }
				NOT = { has_war_with = AUS }
				NOT = { is_ally_with = GER }
			}
		}
		FRA = { give_guarantee = AUS }
	}
}
# Membership bonus follows the federation, including future entrants and departures.
# This grants no factories and cannot farm rewards through repeated faction changes.
GER_nww_refresh_federal_investment = {
	if = {
		limit = {
			is_in_faction = yes
			OR = {
				AND = {
					original_tag = GER
					has_focus_tree = GER_focus
					has_completed_focus = GER_european_federation
					has_government = democratic
					is_faction_leader = yes
				}
				any_other_country = {
					original_tag = GER
					has_focus_tree = GER_focus
					has_completed_focus = GER_european_federation
					has_government = democratic
					is_faction_leader = yes
					is_in_faction_with = PREV
				}
			}
		}
		if = { limit = { NOT = { has_idea = GER_federal_investment } } add_ideas = GER_federal_investment }
	}
	else_if = { limit = { has_idea = GER_federal_investment } remove_ideas = GER_federal_investment }
}
