# MKN owns the flag lifecycle; cosmetic changes never change the gameplay tag.
# Installed references: AUS_scripted_effects.txt, AUS_set_danubian_cosmetic_for_country;
# documentation/effects_documentation.md: set_cosmetic_tag/drop_cosmetic_tag.
MKN_update_national_flag = {
	if = {
		limit = {
			original_tag = MKN
			NOT = { has_country_flag = mkn_batko_departed }
			NOT = { has_country_flag = MKN_duel_participant }
			NOT = { has_cosmetic_tag = MKN_batko }
		}
		set_cosmetic_tag = MKN_batko
	}
	else_if = {
		limit = {
			original_tag = MKN
			has_cosmetic_tag = MKN_batko
			OR = {
				has_country_flag = mkn_batko_departed
				has_country_flag = MKN_duel_participant
			}
		}
		drop_cosmetic_tag = yes
	}
}
