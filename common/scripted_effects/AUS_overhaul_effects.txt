AUS_nww_reduce_bohemian_tension = {
	GER_nww_update_bohemian_influence = yes
	if = {
		limit = { check_variable = { AUS_bohemian_tension > 0 } }
		divide_variable = { AUS_bohemian_german_pressure = AUS_bohemian_tension }
	}
	subtract_from_variable = {
		AUS_bohemian_tension = 2
	}
	clamp_variable = {
		var = AUS_bohemian_tension
		min = 0
		max = 10
	}
	multiply_variable = { AUS_bohemian_german_pressure = AUS_bohemian_tension }
	AUS_nww_update_bohemian_spirit = yes
}
AUS_nww_update_bohemian_spirit = {
	GER_nww_update_bohemian_influence = yes
	set_variable = {
		AUS_bohemian_stability = AUS_bohemian_tension
	}
	multiply_variable = {
		AUS_bohemian_stability = -0.01
	}
	set_variable = {
		AUS_bohemian_output = AUS_bohemian_tension
	}
	multiply_variable = {
		AUS_bohemian_output = -0.01
	}
}
AUS_nww_initialize = {
	if = {
		limit = {
			NOT = {
				has_variable = AUS_bohemian_investments
			}
		}
		set_variable = {
			AUS_bohemian_investments = 0
		}
	}
	if = {
		limit = {
			NOT = {
				has_country_flag = AUS_nww_overhaul_v1
			}
		}
		AUS_nww_rebuild_development = yes
		if = {
			limit = {
				NOT = {
					has_variable = AUS_bohemian_tension
				}
			}
			set_variable = {
				AUS_bohemian_tension = 2
			}
		}
		set_country_flag = AUS_nww_overhaul_v1
	}
	AUS_nww_update_bohemian_spirit = yes
	if = {
		limit = {
			9 = {
				is_owned_by = AUS
			}
			NOT = {
				has_country_flag = AUS_nww_bohemia_settled
			}
			NOT = {
				has_country_flag = AUS_nww_bohemia_seceded
			}
		}
		if = {
			limit = {
				NOT = {
					has_active_mission = AUS_nww_bohemian_drift
				}
			}
			activate_mission = AUS_nww_bohemian_drift
		}
		if = {
			limit = {
				NOT = {
					has_dynamic_modifier = {
						modifier = AUS_nww_bohemian_question
					}
				}
			}
			add_dynamic_modifier = {
				modifier = AUS_nww_bohemian_question
			}
		}
	}
	if = {
		limit = {
			OR = {
				has_country_flag = AUS_nww_bohemia_settled
				has_country_flag = AUS_nww_bohemia_seceded
				NOT = {
					9 = {
						is_owned_by = AUS
					}
				}
			}
		}
		remove_mission = AUS_nww_bohemian_drift
		remove_dynamic_modifier = {
			modifier = AUS_nww_bohemian_question
		}
	}
}
