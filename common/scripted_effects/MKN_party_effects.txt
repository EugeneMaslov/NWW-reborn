# Preserve earned party spirits; switching governments never replays focus rewards.
MKN_prepare_party_revival = {
 if = { limit = { NOT = { has_focus_tree = MKN_focus } } load_focus_tree = { tree = MKN_focus keep_completed = yes } }
 if = {
  limit = {
   OR = {
    has_country_flag = mkn_batko_departed
    has_country_flag = MKN_crisis_legacy_settled
    AND = { is_subject = yes date > 1938.1.1 }
    has_completed_focus = MKN_military_takes_charge
    has_completed_focus = MKN_triumph_of_free_soviets
    has_completed_focus = MKN_peasant_delegates
    has_completed_focus = MKN_emergency_revkom
   }
  }
  set_country_flag = MKN_crisis_enabled
  clr_country_flag = MKN_crisis_legacy_settled
  MKN_resolve_succession_crisis = yes
  if = {
   limit = { NOT = { has_completed_focus = MKN_departure_of_batko } }
   complete_national_focus = MKN_departure_of_batko
  }
  if = {
   limit = { OR = {
    has_completed_focus = MKN_military_takes_charge
    has_completed_focus = MKN_triumph_of_free_soviets
    has_completed_focus = MKN_peasant_delegates
    has_completed_focus = MKN_emergency_revkom
    is_subject = yes
   } }
   set_country_flag = MKN_party_reforms_enabled
  }
 }
}
