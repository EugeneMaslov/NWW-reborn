FRA_select_ai_route = {
	if = {
		limit = {
			has_game_rule = {
				rule = FRA_ai_behavior
				option = REPUBLIC
			}
		}
		set_country_flag = FRA_ai_REPUBLIC
	}
	else_if = {
		limit = {
			has_game_rule = {
				rule = FRA_ai_behavior
				option = MONARCHY
			}
		}
		set_country_flag = FRA_ai_MONARCHY
	}
	else_if = {
		limit = {
			has_game_rule = {
				rule = FRA_ai_behavior
				option = LEAGUES
			}
		}
		set_country_flag = FRA_ai_LEAGUES
	}
	else_if = {
		limit = {
			has_game_rule = {
				rule = FRA_ai_behavior
				option = SOCIAL
			}
		}
		set_country_flag = FRA_ai_SOCIAL
	}
	else_if = {
		limit = {
			has_game_rule = {
				rule = FRA_ai_behavior
				option = COUNCILS
			}
		}
		set_country_flag = FRA_ai_COUNCILS
	}
	else_if = {
		limit = {
			has_game_rule = {
				rule = FRA_ai_behavior
				option = PLAN
			}
		}
		set_country_flag = FRA_ai_PLAN
	}
	else_if = {
		limit = {
			has_game_rule = {
				rule = FRA_ai_behavior
				option = MILITARY
			}
		}
		set_country_flag = FRA_ai_MILITARY
	}
	else_if = {
		limit = {
			has_game_rule = {
				rule = FRA_ai_behavior
				option = DEFAULT
			}
			is_historical_focus_on = yes
		}
		set_country_flag = FRA_ai_REPUBLIC
	}
	else = {
		random_list = {
			20 = {
				set_country_flag = FRA_ai_REPUBLIC
			}
			20 = {
				set_country_flag = FRA_ai_MONARCHY
			}
			20 = {
				set_country_flag = FRA_ai_LEAGUES
			}
			20 = {
				set_country_flag = FRA_ai_SOCIAL
			}
			20 = {
				set_country_flag = FRA_ai_COUNCILS
			}
			20 = {
				set_country_flag = FRA_ai_PLAN
			}
			20 = {
				set_country_flag = FRA_ai_MILITARY
			}
		}
	}
}
