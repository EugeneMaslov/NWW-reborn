# Owner: MKN. La Resistance scopes, per-target reservation, aggregated factory leases.
MKN_begin_expropriation = {
	if = {
		limit = {
			original_tag = MKN
			has_dlc = "La Resistance"
			MKN_valid_expropriation_target = yes
		}
		FROM = {
			set_country_flag = MKN_expropriation_attempted
			if = {
				limit = {
					num_of_civilian_factories > 39
				}
				set_variable = {
					MKN_heist_wealth = 3
				}
			}
			else_if = {
				limit = {
					num_of_civilian_factories > 14
				}
				set_variable = {
					MKN_heist_wealth = 2
				}
			}
			else = {
				set_variable = {
					MKN_heist_wealth = 1
				}
			}
		}
	}
}
MKN_clear_expropriation_factories = {
	if = {
		limit = {
			has_country_flag = MKN_expropriation_factories_active
			NOT = {
				has_variable = MKN_expropriation_factories
			}
		}
		set_variable = {
			MKN_expropriation_factories = 2
		}
	}
	if = {
		limit = {
			check_variable = {
				MKN_expropriation_factories > 0
			}
		}
		# add_offsite_building accepts a signed variable (native CZE weapon sales).
		set_variable = { MKN_expropriation_removal = MKN_expropriation_factories }
		multiply_variable = { MKN_expropriation_removal = -1 }
		add_offsite_building = {
			type = industrial_complex
			level = MKN_expropriation_removal
		}
		clear_variable = MKN_expropriation_removal
	}
	clr_country_flag = MKN_expropriation_factories_active
	clear_variable = MKN_expropriation_factories
}
MKN_expropriation_success = {
	if = {
		limit = {
			MKN_expropriation_outcome_allowed = yes
		}
		FROM = {
			set_country_flag = MKN_expropriation_resolved
			add_timed_idea = { idea = MKN_bank_raid_aftermath days = 180 }
		}
		if = {
			limit = {
				has_country_flag = MKN_expropriation_factories_active
				NOT = {
					has_variable = MKN_expropriation_factories
				}
			}
			set_variable = {
				MKN_expropriation_factories = 2
			}
		}
		if = {
			limit = {
				FROM = {
					check_variable = {
						MKN_heist_wealth = 3
					}
				}
			}
			add_political_power = 75
			add_equipment_to_stockpile = {
				type = infantry_equipment_0
				amount = 750
				producer = ROOT
			}
			add_offsite_building = {
				type = industrial_complex
				level = 3
			}
			add_to_variable = {
				MKN_expropriation_factories = 3
			}
		}
		else_if = {
			limit = {
				FROM = {
					check_variable = {
						MKN_heist_wealth = 2
					}
				}
			}
			add_political_power = 50
			add_equipment_to_stockpile = {
				type = infantry_equipment_0
				amount = 500
				producer = ROOT
			}
			add_offsite_building = {
				type = industrial_complex
				level = 2
			}
			add_to_variable = {
				MKN_expropriation_factories = 2
			}
		}
		else = {
			add_political_power = 25
			add_equipment_to_stockpile = {
				type = infantry_equipment_0
				amount = 300
				producer = ROOT
			}
			add_offsite_building = {
				type = industrial_complex
				level = 1
			}
			add_to_variable = {
				MKN_expropriation_factories = 1
			}
		}
		set_country_flag = MKN_expropriation_factories_active
		if = {
			limit = {
				has_idea = MKN_expropriated_funds
			}
			modify_timed_idea = {
				idea = MKN_expropriated_funds
				days = 90
			}
		}
		else = {
			add_timed_idea = {
				idea = MKN_expropriated_funds
				days = 180
			}
		}
		FROM = {
			country_event = {
				id = mkn_espionage.1
			}
		}
		MKN_dispatch_expropriation_news = yes
	}
}
MKN_expropriation_failure = {
	if = {
		limit = {
			MKN_expropriation_outcome_allowed = yes
		}
		add_political_power = -50
		FROM = {
			set_country_flag = MKN_expropriation_resolved
			if = {
				limit = {
					NOT = {
						has_war_with = ROOT
					}
					NOT = {
						has_wargoal_against = ROOT
					}
				}
				create_wargoal = {
					type = annex_everything
					target = ROOT
					expire = 60
				}
			}
			country_event = {
				id = mkn_espionage.2
			}
		}
	}
}
MKN_raise_black_guard_battalions = {
	if = {
		limit = {
			NOT = {
				has_country_flag = MKN_black_guard_deployed
			}
			1082 = {
				is_owned_and_controlled_by = ROOT
			}
			has_tech = tech_engineers
			has_tech = tech_recon
			has_tech = tech_field_hospital
		}
		set_country_flag = MKN_black_guard_deployed
		load_oob = MKN_black_guard_oob
		army_experience = 20
	}
}
MKN_repair_international_content = {
	# Restore the visible counter of the old fixed lease without re-awarding it.
	if = {
		limit = {
			has_country_flag = MKN_expropriation_factories_active
			has_idea = MKN_expropriated_funds
			NOT = { has_variable = MKN_expropriation_factories }
		}
		set_variable = { MKN_expropriation_factories = 2 }
	}
	if = {
		limit = {
			has_completed_focus = MKN_black_guard_veterans
			NOT = {
				has_completed_focus = MKN_memory_of_atamansha_marusya
			}
		}
		complete_national_focus = MKN_memory_of_atamansha_marusya
		nww_mkn_rebuild_services = yes
	}
	if = {
		limit = {
			has_completed_focus = MKN_memory_of_atamansha_marusya
			NOT = {
				has_idea = MKN_marusyas_legacy
			}
		}
		add_ideas = MKN_marusyas_legacy
	}
	if = {
		limit = {
			has_idea = MKN_marusyas_legacy
			NOT = {
				has_tech = MKN_marusya_assault_training
			}
		}
		set_technology = {
			MKN_marusya_assault_training = 1
		}
	}
	if = {
		limit = {
			has_country_flag = MKN_expropriation_factories_active
			NOT = {
				has_idea = MKN_expropriated_funds
			}
		}
		MKN_clear_expropriation_factories = yes
	}
	if = {
		limit = {
			has_country_flag = MKN_spartacist_factory_active
			NOT = {
				has_idea = MKN_spartacist_engineers
			}
		}
		add_offsite_building = {
			type = arms_factory
			level = -1
		}
		clr_country_flag = MKN_spartacist_factory_active
	}
}
