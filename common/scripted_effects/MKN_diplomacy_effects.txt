# Native relation overrides support volunteers/markets, not declaration bans.
# Scripted diplomacy still checks MKN_protected_diplomatic_relation.
MKN_install_allied_war_guard = {
 if = {
  limit = { tag = UKR is_ai = yes date < 1939.9.1 }
  set_rule = { can_not_declare_war = yes }
  set_country_flag = UKR_mkn_pre1939_war_rule_owned
 }
 else_if = {
  limit = { tag = UKR has_country_flag = UKR_mkn_pre1939_war_rule_owned }
  clear_rule = { can_not_declare_war = yes }
  clr_country_flag = UKR_mkn_pre1939_war_rule_owned
 }
 MKN = {
  if = {
   limit = {
    exists = yes
    SOV = { exists = yes }
    OR = { has_country_flag = MKN_equal_autonomy SOV = { has_country_flag = NWW_SOV_MKN_settlement } }
    NOT = { has_war_with = SOV }
    NOT = { has_non_aggression_pact_with = SOV }
   }
   diplomatic_relation = { country = SOV relation = non_aggression_pact active = yes }
  }
 }
}
UKR_repair_mkn_limited_war = {
	if = {
		limit = {
			has_offensive_war_with = MKN
			MKN = { has_country_flag = mkn_ukr_total_war }
		}
		set_country_flag = UKR_mkn_offensive_limited
	}
	if = {
		limit = { NOT = { has_war_with = MKN } }
		clr_country_flag = UKR_mkn_offensive_limited
	}
}
MKN_close_allied_frontier = {
	if = {
		limit = {
			MKN_ukr_common_bloc = yes
		}
		mkn_ukr_stop_border_battle = yes
		set_country_flag = mkn_ukr_frontier_concluded
		set_country_flag = mkn_ukr_frontier_allied_settlement
		remove_mission = MKN_passive_escalation_drift
		remove_mission = MKN_frontier_year_countdown_mission
		clr_country_flag = mkn_frontier_conflict_active
		clr_country_flag = mkn_ukr_frontier_unlocked
		clr_country_flag = mkn_ukr_war_pending
		clr_country_flag = mkn_ukr_pact_pending
		clr_country_flag = mkn_frontier_year_running
	}
}
