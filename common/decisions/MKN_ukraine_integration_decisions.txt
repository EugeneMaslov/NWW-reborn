MKN_ukrainian_administration_category = {
	MKN_move_capital_to_kyiv = {
		icon = GFX_decision_SOV_place_hq
		allowed = {
			original_tag = MKN
		}
		visible = {
			tag = MKN
			NOT = { UKR = { exists = yes } }
			NOT = { has_country_flag = MKN_capital_moved_to_kyiv }
			202 = { is_capital = no }
		}
		available = {
			MKN_can_move_capital_to_kyiv = yes
		}
		cost = 50
		fire_only_once = yes
		complete_effect = {
			if = {
				limit = { MKN_can_move_capital_to_kyiv = yes }
				set_capital = {
					state = 202
					remember_old_capital = no
				}
				set_country_flag = MKN_capital_moved_to_kyiv
			}
		}
		ai_will_do = { factor = 2 }
	}
}
