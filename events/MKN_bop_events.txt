add_namespace = mkn_bop

# The recipient chooses; completing an MKN focus never forces SOV into an alliance.
country_event = {
	id = mkn_bop.1
	title = mkn_bop.1.t
	desc = mkn_bop.1.d
	picture = GFX_report_event_generic_read_write
	is_triggered_only = yes
	option = {
		name = mkn_bop.1.a
		ai_chance = { factor = 50 }
		trigger = {
			tag = SOV
			has_government = communism
			is_subject = no
			OR = { is_in_faction = no is_faction_leader = yes }
			NOT = { has_war_with = MKN }
			MKN = {
				exists = yes
				has_government = communism
				is_subject = no
				is_in_faction = no
				has_country_flag = MKN_moscow_request_pending
			}
		}
		if = {
			limit = { is_in_faction = no }
			set_rule = { can_create_factions = yes }
			create_faction = MKN_red_defence_pact
		}
		add_to_faction = MKN
		MKN = {
			clr_country_flag = MKN_moscow_request_pending
			set_country_flag = MKN_moscow_alliance
			country_event = { id = mkn_bop.2 days = 1 }
		}
	}
	option = {
		name = mkn_bop.1.b
		ai_chance = { factor = 25 }
		trigger = {
			tag = SOV
			has_government = communism
			is_subject = no
			NOT = { has_war_with = MKN }
			NOT = { is_in_faction_with = MKN }
			MKN = {
				exists = yes
				has_government = communism
				is_subject = no
				is_in_faction = no
				has_country_flag = MKN_moscow_request_pending
			}
		}
		create_wargoal = { type = puppet_wargoal_focus target = MKN }
		MKN = {
			clr_country_flag = MKN_moscow_request_pending
			set_country_flag = MKN_moscow_betrayal
			country_event = { id = mkn_bop.3 days = 1 }
		}
	}
	option = {
		name = mkn_bop.1.c
		ai_chance = { factor = 25 }
		MKN = {
			if = {
				limit = { exists = yes has_country_flag = MKN_moscow_request_pending }
				clr_country_flag = MKN_moscow_request_pending
				country_event = { id = mkn_bop.4 days = 1 }
			}
		}
	}
}
